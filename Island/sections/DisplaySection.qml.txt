import QtQuick
import "../../Common"

// Brillo y saturación por software (shader de pantalla de Hyprland).
// Cada control se puede fijar o quitar del menú por separado.
Column {
    id: root
    property bool brightness: true
    property bool vibrance: true
    spacing: 8
    MSlider {
        width: parent.width
        visible: root.brightness
        icon: "sun"
        from: Brightness.minBrightnessValue; to: 1.0
        // Valor sincronizado desde el singleton Brightness (backlight/DDC/Apple);
        // cualquier cambio externo (atajos, brightnessctl) mueve el deslizador solo.
        value: Brightness.brightness
        valueText: Math.round(value * 100) + "%"
        onMoved: v => Brightness.setBrightness(v)
    }
    MSlider {
        id: satSlider
        width: parent.width
        visible: root.vibrance
        icon: "drop"
        from: 0; to: 2
        value: DisplayState.saturation
        valueText: Math.round(value * 100) + "%"
        iconClickable: true
        onIconClicked: DisplayState.set(1.0, satSlider.value)
        // El brillo ya no pasa por el shader: se conserva el último valor persistido.
        onMoved: v => DisplayState.set(Math.abs(v - 1) < 0.04 ? 1.0 : v, satSlider.value)
    }
}
